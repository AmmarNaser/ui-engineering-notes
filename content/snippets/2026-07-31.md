This snippet provides a customizable button component for a React application that can handle click events and display different styles based on props.

### `CustomButton.js` - Customizable Button Component

```jsx
import React from 'react';
import PropTypes from 'prop-types';

const CustomButton = ({ label, onClick, styleType }) => {
  const buttonStyles = {
    primary: {
      backgroundColor: '#007bff',
      color: '#fff',
    },
    secondary: {
      backgroundColor: '#6c757d',
      color: '#fff',
    },
    success: {
      backgroundColor: '#28a745',
      color: '#fff',
    },
  };

  return (
    <button
      style={buttonStyles[styleType] || buttonStyles.primary}
      onClick={onClick}
    >
      {label}
    </button>
  );
};

CustomButton.propTypes = {
  label: PropTypes.string.isRequired,
  onClick: PropTypes.func.isRequired,
  styleType: PropTypes.string,
};

export default CustomButton;
```

### Usage Example

```jsx
import React from 'react';
import ReactDOM from 'react-dom';
import CustomButton from './CustomButton';

const App = () => {
  const handleClick = () => {
    alert('Button clicked!');
  };

  return (
    <div>
      <CustomButton label="Primary Button" onClick={handleClick} styleType="primary" />
      <CustomButton label="Secondary Button" onClick={handleClick} styleType="secondary" />
      <CustomButton label="Success Button" onClick={handleClick} styleType="success" />
    </div>
  );
};

ReactDOM.render(<App />, document.getElementById('root'));
```
