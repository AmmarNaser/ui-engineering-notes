This code snippet demonstrates a reusable React component for a simple modal dialog. The modal can be easily integrated into any React application, providing a clear way to display content or collect user input.

### `Modal.js` - Reusable Modal Component

```jsx
import React from 'react';
import PropTypes from 'prop-types';
import './Modal.css'; // Include your own CSS for styling

const Modal = ({ isOpen, onClose, title, children }) => {
    if (!isOpen) return null;

    return (
        <div className="modal-overlay">
            <div className="modal-content">
                <h2>{title}</h2>
                <button className="close-button" onClick={onClose}>
                    &times;
                </button>
                <div className="modal-body">
                    {children}
                </div>
            </div>
        </div>
    );
};

Modal.propTypes = {
    isOpen: PropTypes.bool.isRequired,
    onClose: PropTypes.func.isRequired,
    title: PropTypes.string.isRequired,
    children: PropTypes.node.isRequired,
};

export default Modal;
```

### Usage Example

```jsx
import React, { useState } from 'react';
import Modal from './Modal';

const App = () => {
    const [isModalOpen, setModalOpen] = useState(false);

    const toggleModal = () => {
        setModalOpen(!isModalOpen);
    };

    return (
        <div>
            <button onClick={toggleModal}>Open Modal</button>
            <Modal isOpen={isModalOpen} onClose={toggleModal} title="My Modal">
                <p>This is the content of the modal!</p>
            </Modal>
        </div>
    );
};

export default App;
```
