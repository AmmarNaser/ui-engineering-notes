### `Modal.js` - Reusable Modal Component

```jsx
import React, { useEffect } from 'react';
import PropTypes from 'prop-types';
import './Modal.css'; // Assuming you have some basic styles

const Modal = ({ isOpen, onClose, children }) => {
  useEffect(() => {
    const handleEsc = (e) => {
      if (e.key === 'Escape') {
        onClose();
      }
    };
    
    if (isOpen) {
      window.addEventListener('keydown', handleEsc);
    }
    
    return () => {
      window.removeEventListener('keydown', handleEsc);
    };
  }, [isOpen, onClose]);

  if (!isOpen) return null;

  return (
    <div className="modal-overlay">
      <div className="modal-content">
        <button className="close-btn" onClick={onClose}>X</button>
        {children}
      </div>
    </div>
  );
};

Modal.propTypes = {
  isOpen: PropTypes.bool.isRequired,
  onClose: PropTypes.func.isRequired,
  children: PropTypes.node.isRequired,
};

export default Modal;
```

### Usage Example
```jsx
import React, { useState } from 'react';
import Modal from './Modal';

const App = () => {
  const [isModalOpen, setModalOpen] = useState(false);

  const toggleModal = () => {
    setModalOpen(!isModalOpen);
  };

  return (
    <div>
      <h1>My React App</h1>
      <button onClick={toggleModal}>Open Modal</button>
      <Modal isOpen={isModalOpen} onClose={toggleModal}>
        <h2>Modal Title</h2>
        <p>This is the modal content!</p>
      </Modal>
    </div>
  );
};

export default App;
```

### Explanation
- **Modal Component**: The `Modal` component is reusable, taking `isOpen`, `onClose`, and `children` as props to control visibility and render content.
- **Keyboard Accessibility**: The modal listens for the `Escape` key to close, ensuring accessibility for keyboard users.
- **Conditional Rendering**: If the modal is not open, it returns `null`, preventing unnecessary rendering in the DOM.
