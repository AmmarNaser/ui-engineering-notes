This code snippet demonstrates a reusable React component for a modal dialog. It's designed to easily integrate into your application, allowing users to display important information or confirm actions.

### `Modal.js` - Reusable Modal Component

```javascript
import React from 'react';
import './Modal.css'; // Assume you have some basic styles in this file

const Modal = ({ isOpen, onClose, title, children }) => {
  if (!isOpen) return null;

  return (
    <div className="modal-overlay">
      <div className="modal-content">
        <h2>{title}</h2>
        <button className="close-button" onClick={onClose}>Close</button>
        <div className="modal-body">{children}</div>
      </div>
    </div>
  );
};

export default Modal;
```

### Usage Example

```javascript
import React, { useState } from 'react';
import Modal from './Modal';

const App = () => {
  const [isModalOpen, setModalOpen] = useState(false);

  const handleOpenModal = () => setModalOpen(true);
  const handleCloseModal = () => setModalOpen(false);

  return (
    <div>
      <h1>Welcome to the App!</h1>
      <button onClick={handleOpenModal}>Open Modal</button>
      <Modal isOpen={isModalOpen} onClose={handleCloseModal} title="Hello User!">
        <p>This is a reusable modal component.</p>
      </Modal>
    </div>
  );
};

export default App;
```
